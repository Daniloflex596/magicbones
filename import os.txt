import os
import sys
import cv2
import numpy as np

def analyze_video(video_path):
    """
    Analyzes a video file to find the two adjacent frames with the greatest visual difference
    (i.e., the most likely location of a shot change / cut).
    
    Returns:
        tuple: (best_frame_idx, max_diff_value, frame_count, fps, width, height)
               best_frame_idx is the index of the first of the two frames (split point).
    """
    cap = cv2.VideoCapture(video_path)
    if not cap.isOpened():
        print(f"Error: Could not open video file {video_path}", file=sys.stderr)
        sys.exit(1)
        
    fps = cap.get(cv2.CAP_PROP_FPS)
    width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
    total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
    
    if total_frames < 2:
        print("Error: Video has fewer than 2 frames. Cannot compute differences.", file=sys.stderr)
        sys.exit(1)
        
    print(f"Analyzing {total_frames} frames ({width}x{height} @ {fps:.2f} fps)...")
    
    prev_gray = None
    max_diff = -1.0
    best_idx = -1
    
    frame_idx = 0
    while True:
        ret, frame = cap.read()
        if not ret:
            break
            
        # Convert to grayscale for simple, robust structural/histogram comparison
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        
        # Blur slightly to reduce high-frequency noise / compression artifacts
        gray = cv2.GaussianBlur(gray, (5, 5), 0)
        
        if prev_gray is not None:
            # Simple absolute difference of pixel values
            diff = cv2.absdiff(gray, prev_gray)
            mean_diff = np.mean(diff)
            
            # Additional score: structural change via histogram correlation or simple MSE
            if mean_diff > max_diff:
                max_diff = mean_diff
                best_idx = frame_idx - 1 # the frame index prior to change
                
        prev_gray = gray
        frame_idx += 1
        
    cap.release()
    return best_idx, max_diff, total_frames, fps, width, height

def extract_and_split(video_path, split_idx, output_dir="."):
    """
    Extracts the two boundary frames at the split point (split_idx and split_idx + 1)
    and splits the video into two parts:
    - part1: [0, split_idx]
    - part2: [split_idx + 1, end]
    """
    cap = cv2.VideoCapture(video_path)
    if not cap.isOpened():
        print(f"Error: Could not open video file {video_path}", file=sys.stderr)
        sys.exit(1)
        
    fps = cap.get(cv2.CAP_PROP_FPS)
    width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
    total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
    
    # Extract frames
    frame_a = None
    frame_b = None
    
    cap.set(cv2.CAP_PROP_POS_FRAMES, split_idx)
    ret, frame_a = cap.read()
    if not ret:
        print(f"Error: Could not read frame at index {split_idx}", file=sys.stderr)
        sys.exit(1)
        
    ret, frame_b = cap.read()
    if not ret:
        print(f"Error: Could not read frame at index {split_idx + 1}", file=sys.stderr)
        sys.exit(1)
        
    cap.release()
    
    # Save the frames as PNGs
    frame_a_path = os.path.join(output_dir, "frame_a.png")
    frame_b_path = os.path.join(output_dir, "frame_b.png")
    
    cv2.imwrite(frame_a_path, frame_a)
    cv2.imwrite(frame_b_path, frame_b)
    
    # Perform split using ffmpeg for precise/fast split
    split_time_sec = (split_idx + 1) / fps
    
    part1_path = os.path.join(output_dir, "part1.mp4")
    part2_path = os.path.join(output_dir, "part2.mp4")
    
    # Command to extract part1: up to split_time_sec
    cmd_part1 = f'ffmpeg -y -i "{video_path}" -to {split_time_sec} -c:v libx264 -preset superfast -crf 18 -c:a aac "{part1_path}"'
    # Command to extract part2: starting from split_time_sec
    cmd_part2 = f'ffmpeg -y -i "{video_path}" -ss {split_time_sec} -c:v libx264 -preset superfast -crf 18 -c:a aac "{part2_path}"'
    
    print(f"Splitting video at frame {split_idx} (time: {split_time_sec:.3f}s)...")
    os.system(cmd_part1)
    os.system(cmd_part2)
    
    return frame_a_path, frame_b_path, part1_path, part2_path

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: python3 split_utils.py <action> <video_path> [split_idx] [output_dir]")
        sys.exit(1)
        
    action = sys.argv[1]
    
    if action == "analyze":
        v_path = sys.argv[2]
        best_idx, max_diff, frame_count, fps, w, h = analyze_video(v_path)
        print(f"RESULT:split_frame={best_idx}")
        print(f"RESULT:max_diff={max_diff:.4f}")
        print(f"RESULT:frame_count={frame_count}")
        print(f"RESULT:fps={fps}")
        print(f"RESULT:width={w}")
        print(f"RESULT:height={h}")
        print(f"RESULT:time_sec={(best_idx + 1) / fps:.3f}")
        
    elif action == "split":
        v_path = sys.argv[2]
        s_idx = int(sys.argv[3])
        out_dir = sys.argv[4] if len(sys.argv) > 4 else "."
        os.makedirs(out_dir, exist_ok=True)
        
        f_a, f_b, p1, p2 = extract_and_split(v_path, s_idx, out_dir)
        print(f"RESULT:frame_a={os.path.abspath(f_a)}")
        print(f"RESULT:frame_b={os.path.abspath(f_b)}")
        print(f"RESULT:part1={os.path.abspath(p1)}")
        print(f"RESULT:part2={os.path.abspath(p2)}")
